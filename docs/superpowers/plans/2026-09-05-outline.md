# Outline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A left panel listing the whole domain as a tree (Linear-style rows, Shuttle-style collapsed rail), with clicks that select islands across scales, a domain switcher, and the map framed between the outline and the pane.

**Architecture:** `buildOutline` in the core view model walks the island tree once per request behind `GET /api/outline`; the page fetches it beside the snapshot on every version change. `src/web/outline.js` owns the panel: pure helpers (`visibleRows`, `dashClass`, `summaryLine`) are unit-tested, DOM code is checked by hand. The camera takes `{ left, right }` insets. Spec: `docs/superpowers/specs/2026-09-04-outline-design.md`. Prototype to match: https://claude.ai/code/artifact/a251c701-2f8a-4bd2-9e1c-090be3cfa9c0.

**Tech Stack:** Bun, `bun:sqlite`, plain ES modules, canvas, `bun test`.

## Global Constraints

- `bun test` only; `bunx tsc --noEmit` clean. No framework, no bundler, no `innerHTML`.
- `GET /api/outline?domain=<slug>` → `{ version, domain: { slug, name }, harbour, charted, lit, islands: OutlineNode[] }`; `OutlineNode = { slug, name, state, landmarks: { total, unlocked }, harbour, scale, children }` where `state` is the latest verdict's state or `"open"` and `scale` is the parent's slug or null. `charted` and `lit` count over the whole tree. Unknown domain → 404 `{ error }`.
- Outline width 232px expanded, 44px as a rail; the CSS variable `--outline-w` carries it and the top chrome sits at `left: var(--outline-w)`. Not resizable.
- Rows 28px, 13px sans, 6px radius, hover `--panel-2`, selected `rgba(245, 194, 66, .12)`; islets 26px, 12.5px, indented 20px per level; header 44px; section label `Islands`; summary `<n> of <m> charted · <k> lit`; footer `[ outline` and `<m> islands`.
- Rail dashes per top-level island: 14px muted 35% when open and dark; 18px lamp 60% when some landmarks are lit; 22px grass 90% when charted; 26px lamp 100% when selected. Hovering the rail opens the peek card at `left: 52px`, 240px wide, closed on mouse leave.
- Clicking a row selects the island; when its `scale` differs from the current scale the page changes scale first (`await setScale(scale)`) and then selects. Clicking a chevron toggles expansion without selecting. `[` toggles the rail unless focus is in a text field.
- `camera.fit(w, h, { left, right })` and `camera.centreOn(point, w, h, { left, right }, scale)`; `frameSelection` passes the outline's rendered width as `left` and the pane's as `right`.
- Rail state and the expanded set persist in `localStorage` under `outline:<domain>`; every read and write is wrapped in try/catch.
- Copy, verbatim: `Islands`, `<n> of <m> charted · <k> lit`, `[ outline`, `<m> islands`, button titles `collapse to a rail  [` / `expand the outline  [`.

```ts
/** The whole domain as a tree, for the outline panel. Islets are few, so one query per level is fine. */
export function buildOutline(db: Database, domainSlug: string): Outline {
  const domain = requireDomain(db, domainSlug);
  let charted = 0;
  let lit = 0;
  const level = (parent: Island | null): OutlineNode[] => {
    const islands = listIslands(db, domain.id, parent ? parent.id : null);
    const verdicts = latestVerdicts(db, "island", islands.map((i) => i.id));
    const landmarks = listLandmarksByIsland(db, islands.map((i) => i.id));
    return islands.map((island) => {
      const v = verdicts.get(island.id);
      const items = landmarks.get(island.id) ?? [];
      const unlocked = items.filter((l) => l.unlockedAt !== null).length;
      if (v) charted += 1;
      lit += unlocked;
      return {
        slug: island.slug,
        name: island.name,
        state: v ? v.state : "open",
        landmarks: { total: items.length, unlocked },
        harbour: island.harbour === 1,
        scale: parent ? parent.slug : null,
        children: level(island),
      };
    });
  };
  const islands = level(null);
  return {
    version: getViewState(db, domain.id).version,
    domain: { slug: domain.slug, name: domain.name },
    harbour: islands.find((i) => i.harbour)?.slug ?? null,
    charted,
    lit,
    islands,
  };
}
```

```js
    /** The framing that fits the whole world into the area between the insets, centred, with a margin. */
    fit(w, h, { left = 0, right = 0 } = {}) {
      const aw = Math.max(1, w - left - right);
      const scale = (Math.min(aw, h) * (1 - 2 * MARGIN)) / WORLD;
      home = { x: left + (aw - WORLD * scale) / 2, y: (h - WORLD * scale) / 2, scale };
      return home;
    },
    /** The framing at `scale` that puts a world point at the centre of the area between the insets. */
    centreOn(point, w, h, { left = 0, right = 0 } = {}, scale = cam.scale) {
      const aw = Math.max(1, w - left - right);
      return { x: left + aw / 2 - point.x * scale, y: h / 2 - point.y * scale, scale };
    },
```

```js
// The outline: the domain's tree on the left. Pure helpers first (tested), then the DOM.
import { app, setScale, frameSelection } from "./app.js";
import { select } from "./interactions.js";

export const FULL_W = 232;
export const RAIL_W = 44;

/** Whether `slug` is `node` or one of its descendants. */
function holds(node, slug) {
  if (!slug) return false;
  if (node.slug === slug) return true;
  return node.children.some((c) => holds(c, slug));
}

/** The rows to draw: each island, and the islets of islands that are expanded or hold the selection. */
export function visibleRows(islands, expanded, selected, depth = 0) {
  const rows = [];
  for (const node of islands) {
    const hasChildren = node.children.length > 0;
    const open = hasChildren && (expanded.has(node.slug) || holds(node, selected));
    rows.push({ node, depth, hasChildren, open });
    if (open) rows.push(...visibleRows(node.children, expanded, selected, depth + 1));
  }
  return rows;
}

/** The rail dash's class for a top-level island. */
export function dashClass(node, selected) {
  if (node.slug === selected) return "dash on";
  if (node.state !== "open") return "dash done";
  if (node.landmarks.unlocked > 0) return "dash lit";
  return "dash";
}

export function countNodes(islands) {
  return islands.reduce((n, i) => n + 1 + countNodes(i.children), 0);
}

export function summaryLine(outline) {
  return `${outline.charted} of ${countNodes(outline.islands)} charted · ${outline.lit} lit`;
}

// ---- DOM ----

const state = { rail: false, expanded: new Set(), menuOpen: false, domains: null };
let root = null;
let peek = null;

function storageKey() {
  return `outline:${app.domain}`;
}

function load() {
  try {
    const raw = localStorage.getItem(storageKey());
    if (!raw) return;
    const saved = JSON.parse(raw);
    state.rail = Boolean(saved.rail);
    state.expanded = new Set(Array.isArray(saved.expanded) ? saved.expanded : []);
  } catch {
    /* no storage, or a bad value: start fresh */
  }
}

function save() {
  try {
    localStorage.setItem(storageKey(), JSON.stringify({ rail: state.rail, expanded: [...state.expanded] }));
  } catch {
    /* storage unavailable */
  }
}

function el(tag, className, text) {
  const e = document.createElement(tag);
  if (className) e.className = className;
  if (text !== undefined) e.textContent = text;
  return e;
}

function button(label, onClick, className) {
  const b = el("button", className, label);
  b.type = "button";
  b.addEventListener("click", onClick);
  return b;
}

/** Selects an island; if it lives at another scale, the map changes scale first. */
async function go(node) {
  if (node.scale !== app.scale) await setScale(node.scale);
  await select(node.slug);
}

function toggleRail() {
  state.rail = !state.rail;
  document.documentElement.style.setProperty("--outline-w", `${state.rail ? RAIL_W : FULL_W}px`);
  hidePeek();
  save();
  render();
  frameSelection();
}

function toggleExpanded(slug) {
  if (state.expanded.has(slug)) state.expanded.delete(slug);
  else state.expanded.add(slug);
  save();
  render();
}

function listEl(outline) {
  const ul = el("ul", "list");
  for (const row of visibleRows(outline.islands, state.expanded, app.selected)) {
    const li = el("li");
    const r = el("div", `row${row.node.slug === app.selected ? " on" : ""}${row.depth ? " islet" : ""}`);
    r.style.paddingLeft = `${6 + row.depth * 20}px`;
    const tw = button(row.open ? "▾" : "▸", (e) => {
      e.stopPropagation();
      toggleExpanded(row.node.slug);
    }, `tw${row.hasChildren ? "" : " none"}`);
    r.append(tw);
    const lit = row.node.landmarks.unlocked;
    r.append(el("span", `dot ${row.node.state}${row.node.state === "open" && lit ? " lit" : ""}`));
    r.append(el("span", `name${row.node.state === "open" ? " open" : ""}`, row.node.name));
    if (row.node.harbour) r.append(el("span", "anchor", "⚓︎"));
    r.append(el("span", "n", `${lit}/${row.node.landmarks.total}`));
    r.addEventListener("click", () => void go(row.node));
    li.append(r);
    ul.append(li);
  }
  return ul;
}

function showPeek() {
  if (!state.rail || !app.outline) return;
  if (!peek) {
    peek = el("div", "peek");
    peek.id = "peek";
    peek.addEventListener("mouseleave", hidePeek);
    document.body.append(peek);
  }
  peek.hidden = false;
  peek.replaceChildren(listEl(app.outline));
}

function hidePeek() {
  if (peek) peek.hidden = true;
}

async function toggleMenu() {
  state.menuOpen = !state.menuOpen;
  if (state.menuOpen && !state.domains) {
    try {
      const res = await fetch("/api/domains");
      state.domains = (await res.json()).domains;
    } catch {
      state.domains = [];
    }
  }
  render();
}

function headerEl(outline) {
  const head = el("div", "oh");
  const ws = button("", toggleMenu, "ws");
  ws.append(el("span", "name", outline.domain.name), el("span", "chev", "▾"));
  ws.title = "switch domain";
  head.append(ws, el("span", "sp"));
  const tog = button(state.rail ? "»" : "«", toggleRail, "ib");
  tog.title = state.rail ? "expand the outline  [" : "collapse to a rail  [";
  head.append(tog);
  if (state.menuOpen && state.domains) {
    const menu = el("div", "menu");
    for (const d of state.domains) {
      const item = button(d.name, () => {
        const url = new URL(location.href);
        url.searchParams.set("domain", d.slug);
        url.searchParams.delete("scale");
        location.href = url.toString();
      }, `mi${d.slug === app.domain ? " on" : ""}`);
      menu.append(item);
    }
    head.append(menu);
  }
  return head;
}

function railEl(outline) {
  const dashes = el("div", "dashes");
  for (const node of outline.islands) {
    const d = el("div", dashClass(node, app.selected));
    d.title = node.name;
    d.addEventListener("click", () => void go(node));
    dashes.append(d);
  }
  return dashes;
}

function render() {
  if (!root) return;
  const outline = app.outline;
  root.classList.toggle("rail", state.rail);
  if (!outline) {
    root.replaceChildren();
    return;
  }
  const frag = document.createDocumentFragment();
  frag.append(headerEl(outline));
  frag.append(el("div", "sum", summaryLine(outline)));
  frag.append(el("div", "sec", "Islands"));
  frag.append(listEl(outline));
  frag.append(railEl(outline));
  const ft = el("div", "ft");
  const hint = el("span");
  hint.append(el("kbd", null, "["), " outline");
  ft.append(hint, el("span", null, `${countNodes(outline.islands)} islands`));
  frag.append(ft);
  root.replaceChildren(frag);
  if (peek && !peek.hidden) peek.replaceChildren(listEl(outline));
}

export function installOutline() {
  root = document.getElementById("outline");
  load();
  document.documentElement.style.setProperty("--outline-w", `${state.rail ? RAIL_W : FULL_W}px`);
  root.addEventListener("mouseenter", () => showPeek());
  window.addEventListener("keydown", (e) => {
    if (e.key !== "[") return;
    const t = e.target;
    if (t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.isContentEditable)) return;
    e.preventDefault();
    toggleRail();
  });
  app.listeners.add(() => render());
  render();
}
```
