# Themes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Named themes the user picks, `night` (the current look, default) and `chart` (Paper chart), applied to the chrome, the canvas, and the diagrams, remembered per browser.

**Architecture:** Every colour on the page becomes a token on `:root`; a theme is a `[data-theme]` block redefining them. A pure registry (`theme-model.js`) names the themes; `theme.js` stamps the root, stores the choice, pushes seven tokens into the canvas palette, re-initialises mermaid, and fires one `theme` event that the pane and the board answer by rebuilding. The outline's domain menu and the key `t` pick.

**Tech Stack:** Bun, plain ES modules, canvas, mermaid, `bun test`.

Spec: `docs/superpowers/specs/2026-09-07-themes-design.md`. Values below are copied from it.

## Global Constraints

- No framework, no bundler, no `innerHTML`. `bun test` only; `bunx tsc --noEmit` clean.
- The night palette stays exactly as it is. `prefers-color-scheme` is never read.
- The seven canvas tokens (`--sea --line-1 --lamp --reef --sand-fill --grass --snow`) are six-digit hex in every theme.
- Storage key `theme` in `localStorage`; default id `night`.

```js
import { test, expect } from "bun:test";
import { readFileSync } from "node:fs";
import { resolve } from "node:path";

const css = readFileSync(resolve(import.meta.dir, "../../src/web/styles.css"), "utf8");
const SIZES = new Set(["--outline-w", "--pane-w"]);
const PALETTE = ["--sea", "--line-1", "--lamp", "--reef", "--sand-fill", "--grass", "--snow"];

function block(selector) {
  const start = css.indexOf(`${selector} {`);
  if (start < 0) throw new Error(`no block ${selector}`);
  const body = css.slice(start + selector.length + 2, css.indexOf("}", start));
  const out = {};
  for (const m of body.matchAll(/(--[\w-]+)\s*:\s*([^;]+);/g)) out[m[1]] = m[2].trim();
  return out;
}

test("the chart theme defines every literal colour token of :root, and nothing else", () => {
  const root = block(":root");
  const chart = block(':root[data-theme="chart"]');
  const literal = Object.keys(root).filter((k) => !SIZES.has(k) && !root[k].startsWith("var("));
  expect(literal.length).toBeGreaterThan(20);
  expect(Object.keys(chart).filter((k) => !(k in root))).toEqual([]);
  expect(literal.filter((k) => !(k in chart))).toEqual([]);
});

test("the canvas tokens are six-digit hex in both themes", () => {
  for (const sel of [":root", ':root[data-theme="chart"]']) {
    const b = block(sel);
    for (const k of PALETTE) expect(b[k]).toMatch(/^#[0-9A-Fa-f]{6}$/);
  }
});

test("no literal colour survives outside the token blocks", () => {
  const body = css.replace(/:root(\[data-theme="\w+"\])? \{[^}]*\}/g, "");
  expect(body.match(/rgba?\(|#[0-9A-Fa-f]{6}\b/g) ?? []).toEqual([]);
});
```

```js
// The themes the page can wear. A theme is one entry here plus one `:root[data-theme="<id>"]` block in styles.css
// (tests/web/theme-tokens.test.js checks the block covers every colour token). The first entry is the default.
export const THEMES = [
  { id: "night", name: "Night", dark: true },
  { id: "chart", name: "Paper chart", dark: false },
];
export const DEFAULT_THEME = THEMES[0].id;
/** The canvas palette's names → the tokens they are read from (six-digit hex in every theme). */
export const PALETTE_TOKENS = { sea: "--sea", line: "--line-1", lamp: "--lamp", reef: "--reef", sand: "--sand-fill", grass: "--grass", snow: "--snow" };
/** Mermaid's theme variables, by token name, plus the two that are not colours. */
export const MERMAID_TOKENS = {
  background: "--sea", primaryColor: "--panel-2", primaryTextColor: "--text", primaryBorderColor: "--line-03", lineColor: "--lamp",
  secondaryColor: "--panel", tertiaryColor: "--panel", noteBkgColor: "--panel-2", noteTextColor: "--text",
  actorBkg: "--panel-2", actorBorder: "--line-03", actorTextColor: "--text", signalColor: "--lamp", signalTextColor: "--text", edgeLabelBackground: "--sea",
};
export const TOKEN_NAMES = [...new Set([...Object.values(PALETTE_TOKENS), ...Object.values(MERMAID_TOKENS)])];

export function isTheme(id) { return THEMES.some((t) => t.id === id); }
export function storedTheme(raw) { return isTheme(raw) ? raw : DEFAULT_THEME; }
export function nextTheme(id) {
  const i = THEMES.findIndex((t) => t.id === id);
  return THEMES[(i + 1) % THEMES.length].id; // -1 + 1 = 0: unknown → the first
}
export function mermaidVariables(tokens, dark) {
  const out = { darkMode: Boolean(dark), fontFamily: "system-ui, sans-serif", fontSize: "13px" };
  for (const [k, name] of Object.entries(MERMAID_TOKENS)) out[k] = tokens[name];
  return out;
}
```

```js
/** Re-initialises mermaid for a theme and forgets every rendered diagram, so the next append renders in the new colours. */
export function applyMermaidTheme(themeVariables) {
  if (typeof window === "undefined" || !window.mermaid) return false;
  window.mermaid.initialize({
    startOnLoad: false, securityLevel: "strict", theme: "base",
    flowchart: { htmlLabels: false, useMaxWidth: false }, sequence: { useMaxWidth: false },
    themeVariables,
  });
  nodeCache.clear();
  pending.clear();
  mermaidReady = true;
  return true;
}
```

- [ ] **Step 6: `theme.js`**:

```js
// The theme the page wears: stamped on <html>, stored per browser, pushed into the canvas palette and mermaid.
import { THEMES, PALETTE_TOKENS, TOKEN_NAMES, storedTheme, nextTheme } from "./theme-model.js";
import { setPalette } from "./draw.js";
import { applyMermaidTheme } from "./markdown.js";
import { mermaidVariables } from "./theme-model.js";

const KEY = "theme";

export function currentTheme() { return storedTheme(document.documentElement.dataset.theme); }

export function readTokens(names = TOKEN_NAMES) {
  const style = getComputedStyle(document.documentElement);
  const out = {};
  for (const n of names) out[n] = style.getPropertyValue(n).trim();
  return out;
}

export function applyTheme(id) {
  const theme = THEMES.find((t) => t.id === id) ?? THEMES[0];
  document.documentElement.dataset.theme = theme.id;
  try { localStorage.setItem(KEY, theme.id); } catch { /* storage unavailable */ }
  const tokens = readTokens();
  const palette = {};
  for (const [k, name] of Object.entries(PALETTE_TOKENS)) palette[k] = tokens[name];
  setPalette(palette);
  applyMermaidTheme(mermaidVariables(tokens, theme.dark));
  document.dispatchEvent(new CustomEvent("theme", { detail: { id: theme.id } }));
}

export function cycleTheme() { applyTheme(nextTheme(currentTheme())); }

export function installTheme() {
  let raw = null;
  try { raw = localStorage.getItem(KEY); } catch { /* storage unavailable */ }
  applyTheme(storedTheme(raw));
}
```
