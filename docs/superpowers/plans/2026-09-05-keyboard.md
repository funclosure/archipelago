# Keyboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One focused region at a time (outline, map, pane), shown by a lamp line; Linear-style keys dispatched to that region; a cursor in the outline; a `?` cheat sheet; an Escape ladder.

**Architecture:** A pure `keys-model.js` (region cycle, typing check, Escape rung, cursor movement over `visibleRows`, cheat-sheet rows) and a DOM `keys.js` that owns the single window keydown listener, the region state (`body[data-focus]`), the focus line, and the cheat sheet, and dispatches to handlers the three regions register. The existing keydown listeners in `outline.js` and `interactions.js` are removed; their behaviour moves into region handlers. Nothing else changes shape.

**Tech Stack:** plain ES modules, no framework, no `innerHTML`; `bun test` for the pure module.

**Spec:** `docs/superpowers/specs/2026-09-05-keyboard-design.md`

## Global Constraints

- No `innerHTML`; every node via `createElement`/`textContent`/`append`. `bun test` only.
- Regions, exact names and order: `outline`, `map`, `pane`. Initial region `outline`. Tab/Shift-Tab cycle regions and always `preventDefault`, even inside a text field.
- Focus shown by a 1px lamp line: outline right edge (`inset -1px 0 var(--lamp)`), pane left edge (`inset 1px 0 var(--lamp)`), map a 1px bar along the top of the map area (`#map-focus`, `left: var(--outline-w)`, `right: var(--pane-w)` when the pane is shown, else `0`).
- Letters are dead inside text fields; only Escape, Enter, Tab, Cmd-Enter are live there.
- Escape ladder, one rung per press, in this order: cheat sheet open → close it; menu open → close it; typing → blur the field (region stays); outline cursor shown → hide it; selected → deselect; else nothing.
- Outline keys: Down/`j`, Up/`k`, Home, End, Enter, Right, Left, exactly as the spec's table. Cursor never wraps. Cursor seeds to the selected island, else the harbour, else the first row.
- Map keys: Enter (hovered, else nearest the centre of the map area), arrows pan 80px, `+`/`=`/`-` zoom by 1.2 about the centre of the map area, `0` frames home.
- Pane keys: Enter or `/` → answer box; `l` landfall/explore; `n` note; `1`–`4` pick option; `c` confirm reef; `d` focus dispute box; Cmd-Enter sends. Ignored while a turn is in flight except Escape and `?`.
- Cheat sheet: `#keys`, title `Keys`, sections `Everywhere` and the focused region's (`Outline`/`Map`/`Pane`); `?` or Escape closes; it never changes the region.
- Footer keycaps: `[ outline · ? keys`.

```js
// Pure, DOM-free: which region is next, what Escape does, where the outline cursor goes, what the cheat sheet says.

export const REGIONS = ["outline", "map", "pane"];

export function nextRegion(region, backwards = false) {
  const i = Math.max(0, REGIONS.indexOf(region));
  return REGIONS[(i + (backwards ? REGIONS.length - 1 : 1)) % REGIONS.length];
}

/** Whether keystrokes at this target are text entry. */
export function isTyping(target) {
  if (!target) return false;
  const tag = target.tagName;
  return tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || Boolean(target.isContentEditable);
}

/** The one thing Escape does this press, top rung first. */
export function escapeRung({ sheetOpen, menuOpen, typing, cursor, selected }) {
  if (sheetOpen) return "sheet";
  if (menuOpen) return "menu";
  if (typing) return "blur";
  if (cursor) return "cursor";
  if (selected) return "deselect";
  return null;
}

/**
 * The outline cursor after `key`, over the rows `visibleRows` produced.
 * Returns { cursor: slug } to move, { toggle: slug } to expand or collapse, or null for nothing.
 */
export function moveCursor(rows, cursorSlug, key) {
  if (rows.length === 0) return null;
  const i = rows.findIndex((r) => r.node.slug === cursorSlug);
  const at = (j) => ({ cursor: rows[j].node.slug });
  if (i < 0) {
    if (key === "ArrowDown" || key === "j" || key === "ArrowUp" || key === "k" || key === "Home" || key === "End") return at(key === "End" ? rows.length - 1 : 0);
    return null;
  }
  const row = rows[i];
  switch (key) {
    case "ArrowDown":
    case "j":
      return i + 1 < rows.length ? at(i + 1) : null;
    case "ArrowUp":
    case "k":
      return i > 0 ? at(i - 1) : null;
    case "Home":
      return at(0);
    case "End":
      return at(rows.length - 1);
    case "ArrowRight":
      if (!row.hasChildren) return null;
      return row.open ? at(i + 1) : { toggle: row.node.slug };
    case "ArrowLeft": {
      if (row.hasChildren && row.open) return { toggle: row.node.slug };
      for (let j = i - 1; j >= 0; j--) if (rows[j].depth < row.depth) return at(j);
      return null;
    }
    default:
      return null;
  }
}

const EVERYWHERE = [["Tab", "next region"], ["⇧Tab", "previous region"], ["[", "outline"], ["/", "answer"], ["?", "keys"], ["Esc", "back"]];
const REGION_KEYS = {
  outline: { title: "Outline", keys: [["↓ j", "move"], ["↑ k", "move"], ["↵", "select"], ["→ ←", "expand / collapse"], ["Home End", "first / last"]] },
  map: { title: "Map", keys: [["↵", "select"], ["← → ↑ ↓", "pan"], ["+ −", "zoom"], ["0", "home"], ["Esc", "deselect"]] },
  pane: { title: "Pane", keys: [["↵ /", "answer"], ["l", "make landfall"], ["n", "note"], ["1–4", "pick option"], ["c", "confirm reef"], ["d", "dispute"], ["⌘↵", "send"]] },
};

/** The cheat sheet's sections for a region. */
export function cheatSheet(region) {
  return [{ title: "Everywhere", keys: EVERYWHERE }, REGION_KEYS[region] ?? REGION_KEYS.outline];
}
```

```js
// One keydown listener for the page. Exactly one region (outline, map, pane) has focus; keys go to it.
// Global keys (Tab, [, ?, /, Escape, Cmd-Enter) are handled here; everything else is the region's.
import { app } from "./app.js";
import { select } from "./interactions.js";
import { REGIONS, nextRegion, isTyping, escapeRung, cheatSheet } from "./keys-model.js";

const handlers = {}; // region name → handler
let current = "outline";
let sheetOpen = false;

const containers = () => ({
  outline: document.querySelector("#outline .list") ?? document.getElementById("outline"),
  map: document.getElementById("map"),
  pane: document.getElementById("pane"),
});

export function registerRegion(name, handler) {
  handlers[name] = handler;
}

export function region() {
  return current;
}

/** Moves focus to a region: stamps the body, moves DOM focus, tells the regions. */
export function setRegion(name) {
  if (!REGIONS.includes(name)) return;
  const prev = current;
  current = name;
  document.body.dataset.focus = name;
  const c = containers()[name];
  if (c && document.activeElement !== c && !(c.contains(document.activeElement) && isTyping(document.activeElement))) c.focus({ preventScroll: true });
  if (prev !== name) handlers[prev]?.onBlur?.();
  handlers[name]?.onFocus?.();
  if (sheetOpen) renderSheet();
}

function el(tag, className, text) {
  const e = document.createElement(tag);
  if (className) e.className = className;
  if (text !== undefined) e.textContent = text;
  return e;
}

function renderSheet() {
  const box = document.getElementById("keys");
  if (!box) return;
  box.hidden = !sheetOpen;
  box.replaceChildren();
  if (!sheetOpen) return;
  const card = el("div", "keys-card");
  card.append(el("div", "lbl", "Keys"));
  const cols = el("div", "cols");
  for (const section of cheatSheet(current)) {
    const col = el("div", "col");
    col.append(el("div", "sec", section.title));
    for (const [k, label] of section.keys) {
      const row = el("div", "krow");
      row.append(el("kbd", null, k), el("span", null, label));
      col.append(row);
    }
    cols.append(col);
  }
  card.append(cols);
  box.append(card);
}

function toggleSheet() {
  sheetOpen = !sheetOpen;
  renderSheet();
}

function escape(e) {
  const rung = escapeRung({
    sheetOpen,
    menuOpen: handlers.outline?.menuOpen?.() ?? false,
    typing: isTyping(e.target),
    cursor: current === "outline" && (handlers.outline?.hasCursor?.() ?? false),
    selected: Boolean(app.selected),
  });
  if (!rung) return;
  e.preventDefault();
  e.stopImmediatePropagation();
  if (rung === "sheet") toggleSheet();
  else if (rung === "menu") handlers.outline?.closeMenu?.();
  else if (rung === "blur") {
    e.target.blur();
    containers()[current]?.focus({ preventScroll: true });
  } else if (rung === "cursor") handlers.outline?.clearCursor?.();
  else if (rung === "deselect") void select(null);
}

function onKeydown(e) {
  if (e.key === "Tab") {
    e.preventDefault();
    setRegion(nextRegion(current, e.shiftKey));
    return;
  }
  if (e.key === "Escape") {
    escape(e);
    return;
  }
  if (e.key === "Enter" && e.metaKey) {
    if (handlers.pane?.send?.()) e.preventDefault();
    return;
  }
  if (isTyping(e.target)) return; // the field owns its keys
  if (e.metaKey || e.ctrlKey || e.altKey) return; // browser shortcuts stay the browser's
  if (e.key === "?") {
    e.preventDefault();
    toggleSheet();
    return;
  }
  if (e.key === "[") {
    e.preventDefault();
    handlers.outline?.toggleRail?.();
    return;
  }
  if (e.key === "/") {
    e.preventDefault();
    setRegion("pane");
    handlers.pane?.focusAnswer?.();
    return;
  }
  if (sheetOpen) return; // the sheet is modal for region keys
  if (handlers[current]?.onKey?.(e)) e.preventDefault();
}

export function installKeys() {
  window.addEventListener("keydown", onKeydown, { capture: true });
  for (const [name, c] of Object.entries(containers())) {
    c?.addEventListener("mousedown", () => setRegion(name));
  }
  // The outline list is rebuilt on every render; catch its mousedowns from the aside.
  document.getElementById("outline")?.addEventListener("mousedown", () => setRegion("outline"));
  setRegion("outline");
}
```

```js
/** The rows the cursor moves over: the list when expanded, the top-level dashes when railed. */
function cursorRows() {
  if (!app.outline) return [];
  if (state.rail) return app.outline.islands.map((node) => ({ node, depth: 0, hasChildren: false, open: false }));
  return visibleRows(app.outline.islands, state.expanded, app.selected);
}

function seedCursor() {
  const rows = cursorRows();
  if (rows.some((r) => r.node.slug === state.cursor)) return;
  const pick = rows.find((r) => r.node.slug === app.selected) ?? rows.find((r) => r.node.harbour) ?? rows[0];
  state.cursor = pick ? pick.node.slug : null;
}

function scrollCursorIntoView() {
  const row = root.querySelector(".row.cur");
  if (row) row.scrollIntoView({ block: "nearest" });
}

const outlineHandler = {
  onKey(e) {
    if (e.key === "Enter") {
      const row = cursorRows().find((r) => r.node.slug === state.cursor);
      if (!row) return false;
      void go(row.node);
      return true;
    }
    const move = moveCursor(cursorRows(), state.cursor, e.key);
    if (!move) return false;
    if (move.toggle) toggleExpanded(move.toggle);
    else {
      state.cursor = move.cursor;
      render();
      scrollCursorIntoView();
    }
    return true;
  },
  onFocus() {
    seedCursor();
    render();
  },
  onBlur() {
    render(); // the cursor only draws while focused
  },
  menuOpen: () => state.menuOpen,
  closeMenu,
  hasCursor: () => state.cursor !== null,
  clearCursor() {
    state.cursor = null;
    render();
  },
  toggleRail,
};
```

In `installOutline`, after `load();`, add `registerRegion("outline", outlineHandler);`. In `render()`, the cursor class is only meaningful while focused; the CSS already scopes it to `body[data-focus="outline"]`, so no extra logic. Where `toggleExpanded` and `toggleRail` call `render()`, the cursor survives because it is a slug; after `toggleRail` call `seedCursor()` before `render()` so a cursor on an islet falls back to a top-level island when the rail collapses.
